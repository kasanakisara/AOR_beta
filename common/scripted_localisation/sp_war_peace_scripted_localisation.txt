					NOT = { original_tag = DEN }
					NOT = { original_tag = FIC }
					NOT = { original_tag = FIN }
					NOT = { original_tag = ARU }
					NOT = { original_tag = GER }
					NOT = { original_tag = NAU }
					NOT = { original_tag = BAV }
				}

			}
