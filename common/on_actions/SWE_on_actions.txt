on_actions = {

	on_startup = {
		## 开局事件链
		effect = {
			SWE = {
				country_event = { id = Tosca.1 days = 5 }
			}
		}
	}

	on_weekly_SWE = {
		effect = {
			### 门户开放修正
			if = {
				limit = {
					OR = {
						has_idea = SWE_Open_the_door
						has_idea = SWE_Thrid_Chioce
					}
				}

				## 政府支出 -> 生活消费品
				if = {
					limit = { check_variable = { SWE_Government_Spending = 0 compare = less_than_or_equals } }
					set_variable = { SWE_Trade_consumer_goods_factor = 0 }
				}
				else_if = {
					limit = { check_variable = { SWE_Government_Spending = 25 compare = less_than_or_equals } }
					set_variable = { SWE_Trade_consumer_goods_factor = 0.25 }
				}
				else_if = {
					limit = { check_variable = { SWE_Government_Spending = 50 compare = less_than_or_equals } }
					set_variable = { SWE_Trade_consumer_goods_factor = 0.35 }
				}
				else_if = {
					limit = { check_variable = { SWE_Government_Spending = 75 compare = less_than_or_equals } }
					set_variable = { SWE_Trade_consumer_goods_factor = 0.45 }
				}
				else_if = {
					limit = { check_variable = { SWE_Government_Spending = 100 compare = less_than_or_equals } }
					set_variable = { SWE_Trade_consumer_goods_factor = 0.5 }
				}
				else = {
					set_variable = { SWE_Trade_consumer_goods_factor = 0.75 }
				}

				## 政府支出 -> 国际市场开放度
				if = {
					limit = { check_variable = { SWE_Government_Spending = 0 compare = less_than_or_equals } }
					add_to_variable = { SWE_Market_Inclusion = -2.5 }
				}
				else_if = {
					limit = { check_variable = { SWE_Government_Spending = 25 compare = less_than_or_equals } }
					add_to_variable = { SWE_Market_Inclusion = 0 }
				}
				else_if = {
					limit = { check_variable = { SWE_Government_Spending = 50 compare = less_than_or_equals } }
					add_to_variable = { SWE_Market_Inclusion = 2.5 }
				}
				else_if = {
					limit = { check_variable = { SWE_Government_Spending = 75 compare = less_than_or_equals } }
					add_to_variable = { SWE_Market_Inclusion = 5 }
				}
				else_if = {
					limit = { check_variable = { SWE_Government_Spending = 100 compare = less_than_or_equals } }
					add_to_variable = { SWE_Market_Inclusion = 7.5 }
				}
				else = {
					add_to_variable = { SWE_Market_Inclusion = 8.5 }
				}

				## 国际市场开放度上界
				if = {
					limit = { check_variable = { SWE_Market_Inclusion > 100 } }
					set_variable = { SWE_Market_Inclusion = 100 }
				}

				## 国际市场开放度 -> 本土工业发展度
				if = {
					limit = { check_variable = { SWE_Market_Inclusion = 0 compare = less_than_or_equals } }
					add_to_variable = { SWE_Industry_Development = -2.5 }
				}
				else_if = {
					limit = { check_variable = { SWE_Market_Inclusion = 25 compare = less_than_or_equals } }
					add_to_variable = { SWE_Industry_Development = 0 }
				}
				else_if = {
					limit = { check_variable = { SWE_Market_Inclusion = 50 compare = less_than_or_equals } }
					add_to_variable = { SWE_Industry_Development = 2.5 }
				}
				else_if = {
					limit = { check_variable = { SWE_Market_Inclusion = 75 compare = less_than_or_equals } }
					add_to_variable = { SWE_Industry_Development = 5 }
				}
				else_if = {
					limit = { check_variable = { SWE_Market_Inclusion = 100 compare = less_than_or_equals } }
					add_to_variable = { SWE_Industry_Development = 7.5 }
				}

				## 本土工业发展度上下界
				if = {
					limit = { check_variable = { SWE_Industry_Development > 100 } }
					set_variable = { SWE_Industry_Development = 100 }
				}
				if = {
					limit = { check_variable = { SWE_Industry_Development < 0 } }
					set_variable = { SWE_Industry_Development = 0 }
				}

				## 国际市场开放度 -> 建造速度
				if = {
					limit = { check_variable = { SWE_Market_Inclusion = 0 compare = less_than_or_equals } }
					set_variable = { SWE_Trade_construction_speed = -0.05 }
				}
				else_if = {
					limit = { check_variable = { SWE_Market_Inclusion = 25 compare = less_than_or_equals } }
					set_variable = { SWE_Trade_construction_speed = 0 }
				}
				else_if = {
					limit = { check_variable = { SWE_Market_Inclusion = 50 compare = less_than_or_equals } }
					set_variable = { SWE_Trade_construction_speed = 0.15 }
				}
				else_if = {
					limit = { check_variable = { SWE_Market_Inclusion = 75 compare = less_than_or_equals } }
					set_variable = { SWE_Trade_construction_speed = 0.25 }
				}
				else_if = {
					limit = { check_variable = { SWE_Market_Inclusion = 100 compare = less_than_or_equals } }
					set_variable = { SWE_Trade_construction_speed = 0.45 }
				}
				else = {
					set_variable = { SWE_Trade_construction_speed = 0.5 }
				}
			}
		}
	}

	on_monthly_SWE = {
		effect = {
			#人民之家
			if = {
				limit = {
					has_idea = SWE_Folkhemmet
				}
				add_political_power = -50
			}

			## 门户开放
			if = {
				limit = {
					OR = {
						has_idea = SWE_Open_the_door
						has_idea = SWE_Third_Chioce
					}
				}

				## 检测本土工业发展度
				if = {
					limit = { check_variable = { SWE_Industry_Development = 25 compare = less_than_or_equals } }
					random_owned_controlled_state = {
						limit = { is_core_of = SWE }
						add_extra_state_shared_building_slots = 1
						add_building_construction = {
							type = industrial_complex
							level = 1
							instant_build = yes 
						}
					}
				}

				else_if = {
					limit = { check_variable = { SWE_Industry_Development = 75 compare = less_than_or_equals } }
					random_owned_controlled_state = {
						limit = { is_core_of = SWE }
						add_extra_state_shared_building_slots = 2
						add_building_construction = {
							type = industrial_complex
							level = 2
							instant_build = yes 
						}
					}
				}

				else = {
					random_owned_controlled_state = {
						limit = { is_core_of = SWE }
						add_extra_state_shared_building_slots = 3
						add_building_construction = {
							type = industrial_complex
							level = 3
							instant_build = yes 
						}
					}
				}


				country_event = { id = SWE.25 }
				add_to_variable = { SWE_Industry_Development = -15 }
			}
		}
	}

	on_daily_SWE = {
		effect = {
			###### 挪威独立度增长决议 ######
			if = {
				limit = {
					has_country_flag = SWE_Norweign_Independence_flag
				}

				#增加数值
				set_variable = { SWE_Norweign_Independence_add = 0 }
				if = {
					limit = {
						has_country_flag = SWE_GUI_1 #哈康初代
					}
					add_to_variable = { SWE_Norweign_Independence_add = 2 }
				}
				if = {
					limit = {
						OR = {
							has_country_flag = SWE_GUI_2 #紧急状态
							has_country_flag = SWE_GUI_3 #鲁普雷希特初期
						}
					}
					add_to_variable = { SWE_Norweign_Independence_add = 3.5 }
				}

				#民族精神修正
				if = { 
					limit = {
						has_idea = SWE_Eighty_Seventh_Orders
					}
					add_to_variable = { SWE_Norweign_Independence_add = -2 }
				}
				if = { 
					limit = {
						has_idea = A_lingering_alliance
					}
					add_to_variable = { SWE_Norweign_Independence_add = 2.5 }
				}
				if = { 
					limit = {
						has_idea = SWE_Left_Threat
					}
					add_to_variable = { SWE_Norweign_Independence_add = 1 }
				}
				if = { 
					limit = {
						has_idea = SWE_Left_Threat_I
					}
					add_to_variable = { SWE_Norweign_Independence_add = 0.75 }
				}
				if = { 
					limit = {
						has_idea = SWE_Left_Threat_II
					}
					add_to_variable = { SWE_Norweign_Independence_add = 0.5 }
				}
				if = { 
					limit = {
						has_idea = SWE_Antisocialistisk_Speciallagen
					}
					add_to_variable = { SWE_Norweign_Independence_add = -3.5 }
				}
				if = { 
					limit = {
						has_idea = SWE_All_People_are_King
					}
					add_to_variable = { SWE_Norweign_Independence_add = -1 }
				}
				if = { 
					limit = {
						has_idea = SWE_Beyond_Nationalism
					}
					add_to_variable = { SWE_Norweign_Independence_add = -2.5 }
				}
				if = { 
					limit = {
						has_idea = SWE_Beyond_Nation
					}
					add_to_variable = { SWE_Norweign_Independence_add = -3 }
				}


				#马克思主义支持率对独立度修正
				SWE_Marxism_effect_Norway_Independence_check = yes
				
				#增长下限
				if = {
					limit = {
						check_variable = { SWE_Norweign_Independence_add = 0 compare = less_than_or_equals }
					}
					set_variable = { SWE_Norweign_Independence_add = 0.25 }
				}

				if = {
					limit = {
						has_country_flag = SWE_GUI_4 #哈康抑制期
					}
					set_variable = { SWE_Norweign_Independence_add = 0 }
				}

				#数值更新
				add_to_variable = { SWE_Norweign_Independence_value = SWE_Norweign_Independence_add }

				#更新gui
				SWE_update_ui_effect = yes


				#修正更新
				SWE_Norweign_Independence_value_check = yes


				#上下限
				if = {
					limit = {
						check_variable = { SWE_Norweign_Independence_value = 0 compare = less_than_or_equals }
					}
					set_variable = { SWE_Norweign_Independence_value = 0 }
				}
				if = {
					limit = {
						check_variable = { SWE_Norweign_Independence_value > 100 }
					}
					set_variable = { SWE_Norweign_Independence_value = 100 }
				}

				#判定是否触发挪威独立事件
				if = {
					limit = {
						check_variable = { SWE_Norweign_Independence_value = 100 }
					}

					if = {
						limit = {
							has_country_flag = SWE_GUI_1
						}
						set_variable = { SWE_Norweign_Independence_value = 0 }
						remove_dynamic_modifiers = SWE_Norweign_Indepedence_modifier
						country_event = { id = SWE.11 }
					}
				}

			}

			## Группа Крови на рукаве


			## 联邦一梦
			if = {
				limit = {
					has_idea = SWE_The_Union_Dream
				}

				set_variable = { SWE_Union_progress_value_add = 0 }

				## 市场包容度 -> 增长度

				if = {
					limit = {
						check_variable = { SWE_Market_Inclusion = 15 compare = less_than_or_more }
					}
					add_to_variable = { SWE_Union_progress_value_add = -0.25 }
				}
				else_if = {
					limit = {
						check_variable = { SWE_Market_Inclusion = 25 compare = less_than_or_more }
					}
					add_to_variable = { SWE_Union_progress_value_add = 0 }
				}
				else_if = {
					limit = {
						check_variable = { SWE_Market_Inclusion = 45 compare = less_than_or_more }
					}
					add_to_variable = { SWE_Union_progress_value_add = 0.25 }
				}
				else_if = {
					limit = {
						check_variable = { SWE_Market_Inclusion = 45 compare = less_than_or_more }
					}
					add_to_variable = { SWE_Union_progress_value_add = 0.25 }
				}
				else_if = {
					limit = {
						check_variable = { SWE_Market_Inclusion = 65 compare = less_than_or_more }
					}
					add_to_variable = { SWE_Union_progress_value_add = 0.5 }
				}
				else_if = {
					limit = {
						check_variable = { SWE_Market_Inclusion = 75 compare = less_than_or_more }
					}
					add_to_variable = { SWE_Union_progress_value_add = 0.65 }
				}
				else_if = {
					limit = {
						check_variable = { SWE_Market_Inclusion = 85 compare = less_than_or_more }
					}
					add_to_variable = { SWE_Union_progress_value_add = 0.75 }
				}
				else = {
					add_to_variable = { SWE_Union_progress_value_add = 1 }
				}

				#加和
				add_to_variable = { SWE_Union_progress_value = var:SWE_Union_progress_value_add }

				#判断是否进步
				if = {
					limit = {
						check_variable = { SWE_Union_progress_value = 100 compare = more_than_or_equals }
					}
					add_to_variable = { SWE_Union_Stage = 1 }
					add_to_variable = { SWE_Union_progress_value = -100 }
				}

			}

		}

	}
}
