on_action = {
	on_daily_SWE = {
		effect = {
			#每日减少的议会态度
			if = {
				limit = {
					check_variable = {SWE_Sweden_Parliament_Attitude < SWE_Stortinget_Attitude}
				}
				subtract_from_variable = {
					var = SWE_NationalUnity
					value = SWE_Sweden_Parliament_Attitude
				}
				add_to_variable = {
					var = SWE_NationalUnity
					value = SWE_Stortinget_Attitude
				}
			}
			else = {
				subtract_from_variable = {
					var = SWE_NationalUnity
					value = SWE_Stortinget_Attitude
				}
				add_to_variable = {
					var = SWE_NationalUnity
					value = SWE_Sweden_Parliament_Attitude
				}
			}

			if = {
				limit = {
					check_variable = {SWE_Sweden_Parliament_Attitude < 0}
				}
				add_to_variable = {
					var = SWE_Sweden_Parliament_Attuitude
					value = 20
				}
			}

			subtract_from_variable = {
				var = SWE_Stortinget_Attitude
				value = 7.5
			}

			add_to_variable = {
				var = SWE_Left_Threat_value
				value = 0.01
			}

			if = {
				limit = {
					check_variable = { SWE_Left_Threat_value > 0.3 }
					has_ideas = SWE_Left_Threat
				}
				swap_ideas = {
					remove_idea = SWE_Left_Threat
					add_idea = SWE_Left_Threat_II
				}
			}

			if = {
				limit = {
					check_variable = { SWE_Left_Threat_value > 0.5 }
					has_ideas = SWE_Left_Threat_II
				}
				swap_ideas = {
					remove_idea = SWE_Left_Threat_II
					add_idea = SWE_Left_Threat_III
				}
			}

			if = {
				limit = {
					check_variable = { SWE_Left_Threat_value > 0.75 }
					has_ideas = SWE_Left_Threat_III
				}
				remove_idea = SWE_Left_Threat_III
				country_event = {id = SWE.25}
			}

			if = {
				limit = {
					Marxism > 0.5
				}
				add_to_variable = {
					var = SWE_Left_Threat_value
					value = 0.05
				}
			}
		}
	}
}
