scripted_gui = {
	AOR_leaders_button = {
		window_name = "AOR_leaders_button"
		context_type = country_context
		visible = {
			always = yes
		}
		parent_window_name = "countrypoliticsview"
		effects = {
			AOR_leaders_button_1_click ={
				if = {
					limit = {
						NOT = {
							has_country_flag = open_leaders_window
						}
					}
					set_country_flag = open_leaders_window
				}
				else = {
					clr_country_flag = open_leaders_window
				}
			}
		}
	}

	AOR_leaders_window = {
		window_name = "AOR_leaders_window"
		context_type = player_context

		visible = {
			has_country_flag = open_leaders_window
		}
	}
}
