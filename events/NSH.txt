#石勒苏益格-荷尔施泰因公投
country_event = {
	id = NSH.2
	title = NSH.2.t
	desc = NSH.2.d
	picture = GFX_events_NSH_2
	is_triggered_only = yes	
	fire_only_once = yes

	option = {  #石勒苏益格-荷尔施泰因公投入德
		name = NSH.2.a
		DEN = { country_event = { id = NSH.3 } }
		news_event = { id = AOR_news.211 }

		GER = {
			add_state_core = 58
			add_state_core = 1043
		}	
		SHL = {
			add_state_core = 58
			add_state_core = 1043
		}	
		58 = { set_demilitarized_zone = yes }
		1043 = { set_demilitarized_zone = yes }

		ai_chance = {
			base = 70
		}
    }
	option = { #石勒苏益格-荷尔施泰因公投入丹 （这不太可能）
		name = NSH.2.b 
		news_event = { id = AOR_news.212 }
		SHL = { country_event = { id = NSH.4 } }

		DEN = {
			add_state_core = 58
			add_state_core = 1043
		}
		if = {
			limit ={
				DEN = {
					controls_state = 839
				}
			}
			839 = { set_demilitarized_zone = yes }
		}
		58 = { set_demilitarized_zone = yes }
		1043 = { set_demilitarized_zone = yes }

		ai_chance = {
			base = 5
		}
    }
	option = { #石勒苏益格-荷尔施泰因公投作废
	name = NSH.2.c
	news_event = { id = AOR_news.213 }

	ai_chance = {
		base = 25
	}
}
}

		}
    }
