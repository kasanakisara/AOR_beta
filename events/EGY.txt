	id = EGY.1
	title = EGY.1.t
	desc = EGY.1.d
	picture = GFX_events_EGY_1

	trigger = {
	    date > 1922.1.1
		tag = EGY
	}
	
	mean_time_to_happen = { day = 1}
	
	fire_only_once = yes

    #那就这样吧  
	immediate = {
		retire_country_leader = yes
		set_politics = {ruling_party = ISL_democratic}
		create_country_leader = {
		name = "Saad Zaghloul"
		desc = "POLITICS_Saad_Zaghloul_DESC"
		picture = "gfx/leaders/EGY/Zaghloul.png"
		expire = "1945.1.1"
		ideology = constitutionalism
		}
		set_popularities = {
				Absolutism = 50
				CON_democratic = 25
				ISL_democratic = 20
				FRE_democratic = 5
				}
	}	
	option = {
		name = EGY.1.a
		
    }
}
